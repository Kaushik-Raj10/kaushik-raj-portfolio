const Trefoil = (() => {
  // A swept tube along a (2,3) torus knot, lit and projected into a 2D canvas.
  const TAU = Math.PI * 2;
  const REST = { pitch: -0.32, yaw: 0.33, roll: -0.48 };

  const colors = {
    copper: [[56, 21, 14], [143, 47, 20], [235, 88, 32], [255, 152, 77], [255, 222, 161]],
    iris: [[35, 23, 55], [89, 58, 136], [155, 117, 225], [204, 175, 255], [247, 229, 255]],
    silver: [[27, 34, 31], [66, 84, 78], [140, 163, 154], [202, 217, 207], [248, 250, 237]]
  };

  function palette(stops) {
    return Array.from({ length: 128 }, (_, i) => {
      const t = i / 127 * (stops.length - 1);
      const a = Math.min(stops.length - 2, Math.floor(t));
      const rgb = stops[a].map((c, j) => Math.round(c + (stops[a + 1][j] - c) * (t - a)));
      return 'rgb(' + rgb.join(',') + ')';
    });
  }
  const materials = Object.fromEntries(Object.entries(colors).map(([key, stops]) => [
    key, { rib: palette(stops), core: palette(stops.map((rgb, i) => rgb.map(c => Math.round(c * (.18 + i * .038))))) }
  ]));

  function makeKnot(rings = 192, sides = 32) {
    const count = rings * sides;
    const vertices = new Float32Array(count * 6);
    const screen = new Float32Array(count * 6);
    const faces = [];
    for (let i = 0; i < rings; i++) {
      const t = TAU * i / rings;
      const a = 1.38 + .66 * Math.cos(3 * t);
      const dr = -1.98 * Math.sin(3 * t);
      const c = Math.cos(2 * t), s = Math.sin(2 * t);
      const center = [a * c, a * s, .81 * Math.sin(3 * t)];
      const tx = dr * c - 2 * a * s, ty = dr * s + 2 * a * c, tz = 2.43 * Math.cos(3 * t);
      const L = Math.hypot(tx, ty, tz), flat = Math.hypot(tx, ty);
      const ax = -ty / flat, ay = tx / flat;
      const bx = -tz / L * ay, by = tz / L * ax, bz = (tx * ay - ty * ax) / L;
      const radius = .465 + .027 * Math.cos(3 * t + .65);
      for (let j = 0; j < sides; j++) {
        const theta = TAU * j / sides;
        const u = Math.cos(theta), v = Math.sin(theta);
        const nx = ax * u + bx * v, ny = ay * u + by * v, nz = bz * v;
        const k = (i * sides + j) * 6;
        vertices.set([center[0] + radius * nx, center[1] + radius * ny, center[2] + radius * nz, nx, ny, nz], k);
        const ni = (i + 1) % rings, nj = (j + 1) % sides;
        faces.push({ a: k, b: (i * sides + nj) * 6, c: (ni * sides + nj) * 6, d: (ni * sides + j) * 6, z: 0 });
      }
    }
    return { vertices, screen, faces, visible: [] };
  }

  function drawKnot(ctx, width, height, mesh, pose = REST, material = 'copper') {
    const { vertices, screen, faces, visible } = mesh;
    const cp = Math.cos(pose.pitch), sp = Math.sin(pose.pitch);
    const cy = Math.cos(pose.yaw), sy = Math.sin(pose.yaw);
    const cr = Math.cos(pose.roll), sr = Math.sin(pose.roll);
    const m = [cr*cy-sr*sp*sy, -sr*cp, cr*sy+sr*sp*cy,
               sr*cy+cr*sp*sy, cr*cp, sr*sy-cr*sp*cy,
               -cp*sy, sp, cp*cy];
    let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
    for (let k = 0; k < vertices.length; k += 6) {
      const x=vertices[k], y=vertices[k+1], z=vertices[k+2];
      const nx=vertices[k+3], ny=vertices[k+4], nz=vertices[k+5];
      const depth=m[6]*x+m[7]*y+m[8]*z;
      const p=8.8/(8.8-depth);
      const X=(m[0]*x+m[1]*y+m[2]*z)*p, Y=-(m[3]*x+m[4]*y+m[5]*z)*p;
      screen[k]=X; screen[k+1]=Y; screen[k+2]=depth;
      screen[k+3]=m[0]*nx+m[1]*ny+m[2]*nz;
      screen[k+4]=m[3]*nx+m[4]*ny+m[5]*nz;
      screen[k+5]=m[6]*nx+m[7]*ny+m[8]*nz;
      minX=Math.min(minX,X);maxX=Math.max(maxX,X);
      minY=Math.min(minY,Y);maxY=Math.max(maxY,Y);
    }
    const scale=Math.min(width*.92/(maxX-minX),height*.95/(maxY-minY));
    const ox=(width-(maxX+minX)*scale)/2, oy=(height-(maxY+minY)*scale)/2;
    for(let k=0;k<screen.length;k+=6){screen[k]=screen[k]*scale+ox;screen[k+1]=screen[k+1]*scale+oy;}
    visible.length=0;
    for(const face of faces){
      const {a,b,c,d}=face;
      const facing=(screen[b]-screen[a])*(screen[c+1]-screen[a+1])-(screen[b+1]-screen[a+1])*(screen[c]-screen[a]);
      if(facing>0) continue;
      face.z=(screen[a+2]+screen[b+2]+screen[c+2]+screen[d+2])*.25;
      visible.push(face);
    }
    visible.sort((a,b)=>a.z-b.z);
    ctx.clearRect(0,0,width,height);
    const paint=materials[material] || materials.copper;
    ctx.lineJoin='round';
    for(const {a,b,c,d,z} of visible){
      const nx=(screen[a+3]+screen[c+3])*.5, ny=(screen[a+4]+screen[c+4])*.5, nz=(screen[a+5]+screen[c+5])*.5;
      const diffuse=Math.max(0,-.49*nx+.68*ny+.55*nz);
      const reflection=Math.max(0,-.27*nx+.4*ny+.875*nz);
      const rim=Math.pow(1-Math.max(0,nz),2);
      const light=Math.min(1,Math.max(0,.18+.52*diffuse+.27*Math.pow(reflection,12)+.14*rim+.038*z));
      const tone=Math.round(light*127);
      const ax=screen[a],ay=screen[a+1],bx=screen[b],by=screen[b+1],cx=screen[c],cy=screen[c+1],dx=screen[d],dy=screen[d+1];
      ctx.beginPath();ctx.moveTo(ax,ay);ctx.lineTo(bx,by);ctx.lineTo(cx,cy);ctx.lineTo(dx,dy);ctx.closePath();
      ctx.fillStyle=paint.core[tone];ctx.fill();
      // Fill the raised part of each copper rib. Its projected width narrows naturally at crossings.
      const f=.37;
      ctx.beginPath();ctx.moveTo(ax,ay);ctx.lineTo(bx,by);
      ctx.lineTo(bx+(cx-bx)*f,by+(cy-by)*f);ctx.lineTo(ax+(dx-ax)*f,ay+(dy-ay)*f);ctx.closePath();
      ctx.fillStyle=paint.rib[tone];ctx.fill();
    }
  }

  function mountKnot(canvas, shouldMove) {

    if(!ctx) return;

    let material='copper', active=true, drag=null, last=0, elapsed=0, yaw=0, pitch=0, frame=0, dirty=true;
    const still=canvas.previousElementSibling;
    function request(){if(!frame)frame=requestAnimationFrame(render);}
    function render(now){
      frame=0;
      if(!active||document.hidden){last=0;return;}
      const running=shouldMove()&&!drag;
      if(running && last && now-last<36 && !dirty){request();return;}
      const step=last?Math.min((now-last)/1000,.09):0;
      if(running)elapsed+=step;
      last=now;
      if(running||dirty){
        const rect=canvas.getBoundingClientRect();
        if(!rect.width||!rect.height)return;
        const dpr=Math.min(devicePixelRatio||1,1.65);
        const w=Math.round(rect.width*dpr),h=Math.round(rect.height*dpr);
        if(canvas.width!==w||canvas.height!==h){canvas.width=w;canvas.height=h;}
        pose.pitch=REST.pitch+pitch+.11*Math.sin(elapsed*.27);
        pose.yaw=REST.yaw+yaw+.6*Math.sin(elapsed*.17);
        pose.roll=REST.roll+.065*Math.sin(elapsed*.25);
        drawKnot(ctx,w,h,mesh,pose,material);
        still.style.visibility='hidden';
        dirty=false;
      }
      if(running)request();
    }
    const refresh=()=>{dirty=true;last=0;request();};
    new ResizeObserver(refresh).observe(canvas);
    new IntersectionObserver(([entry])=>{active=entry.isIntersecting;if(active)refresh();},{rootMargin:'35px'}).observe(canvas);
    document.addEventListener('visibilitychange',refresh);
    window.addEventListener('portfolio-motion',refresh);
    canvas.addEventListener('pointerdown',e=>{
      if(e.button!==0)return;
      drag={x:e.clientX,y:e.clientY,yaw,pitch};

    });

    });
    function release(){if(!drag)return;drag=null;canvas.classList.remove('dragging');refresh();}
    canvas.addEventListener('pointerup',release);canvas.addEventListener('pointercancel',release);canvas.addEventListener('lostpointercapture',release);
    canvas.addEventListener('keydown',e=>{

      if(e.key==='Home'){elapsed=0;yaw=0;pitch=0;}

'use strict';

// Each snapshot stores the array and the counters for exactly one visible step.
function createSortTrace(input, method) {
  const values = input.slice();
  const frames = [];
  let comparisons = 0;
  let swaps = 0;

  function record(kind, pair, message) {
    frames.push({ values: values.slice(), kind, pair, message, comparisons, swaps });
  }

  function compare(left, right) {
    comparisons++;
    record('compare', [left, right], 'Compare ' + values[left] + ' and ' + values[right] + '.');
    return values[left] - values[right];
  }

  function swap(left, right) {
    const a = values[left];
    const b = values[right];
    [values[left], values[right]] = [b, a];
    swaps++;
    record('swap', [left, right], 'Swap ' + a + ' and ' + b + '.');
  }

  record('start', [], 'Ready to sort this array.');
  if (method === 'bubble') {
    for (let end = values.length - 1; end > 0; end--) {
      let changed = false;
      for (let left = 0; left < end; left++) {
        if (compare(left, left + 1) > 0) {
          swap(left, left + 1);
          changed = true;
        }
      }
      if (!changed) break;
    }
  } else if (method === 'insertion') {
    for (let next = 1; next < values.length; next++) {
      let position = next;
      while (position > 0 && compare(position - 1, position) > 0) {
        swap(position - 1, position);
        position--;
      }
    }
  } else if (method === 'selection') {
    for (let first = 0; first < values.length - 1; first++) {
      let smallest = first;
      for (let cursor = first + 1; cursor < values.length; cursor++) {
        if (compare(cursor, smallest) < 0) smallest = cursor;
      }
      if (smallest !== first) swap(first, smallest);
    }
  } else {
    throw new Error('Unknown sorting algorithm');
  }
  record('done', [], 'In order! ' + comparisons + ' comparisons and ' + swaps + ' swaps.');
  return frames;
}

(() => {
  const numbers = [7, 3, 11, 2, 9, 5, 12, 1, 10, 4, 8, 6];
  const descriptions = {
    bubble: 'Compare neighbors and swap the pair when the larger value is on the left. Each pass carries a large value to the end.',
    insertion: 'Start with one value in order. Move each new value left past its larger neighbors until it fits.',
    selection: 'Find the smallest value in the remaining group and swap it into the next open position.'
  };
  const byId = (id) => document.getElementById(id);
  const chart = byId('bars');
  const method = byId('algorithm');
  const speed = byId('speed');
  const play = byId('play');
  const step = byId('step');
  const announcement = byId('sort-announcement');

  const bars = numbers.map(() => {
    const cell = document.createElement('div');
    const column = document.createElement('span');
    const label = document.createElement('span');
    cell.className = 'bar-cell';
    column.className = 'bar-column';
    label.className = 'bar-number';
    cell.append(column, label);
    chart.append(cell);
    return { cell, column, label };
  });

  let trace = [];
  let index = 0;
  let timer = null;

  function paint() {
    const frame = trace[index];
    frame.values.forEach((value, position) => {
      const bar = bars[position];
      bar.column.style.height = (12 + value * 6.6) + '%';
      const isActive = frame.pair.includes(position);
      bar.cell.dataset.state = frame.kind === 'done' ? 'done' : isActive ? frame.kind : 'idle';
      bar.label.textContent = value;
    });
    chart.setAttribute('aria-label', 'Array from left to right: ' + frame.values.join(', '));
    byId('comparisons').textContent = frame.comparisons;
    byId('swaps').textContent = frame.swaps;
    byId('step-number').textContent = index + ' / ' + (trace.length - 1);
    byId('event-text').textContent = frame.message;
    step.disabled = frame.kind === 'done';
    play.textContent = timer !== null ? 'Pause' : frame.kind === 'done' ? 'Replay' : 'Play';
  }

  function stop() {
    clearInterval(timer);
    timer = null;
  }

  function advance(manual = false) {
    if (index < trace.length - 1) index++;
    if (index === trace.length - 1) stop();
    paint();
    if (manual || index === trace.length - 1) announcement.textContent = trace[index].message;
  }

  function start() {
    stop();
    if (index === trace.length - 1) index = 0;
    timer = setInterval(advance, Number(speed.value));
    paint();
  }

  function reset() {
    stop();
    index = 0;
    trace = createSortTrace(numbers, method.value);
    byId('method-name').textContent = method.selectedOptions[0].textContent;
    byId('method-note').textContent = descriptions[method.value];
    announcement.textContent = '';
    paint();
  }

  play.addEventListener('click', () => {
    if (timer === null) start();
    else { stop(); paint(); }
  });
  step.addEventListener('click', () => { stop(); advance(true); });
  byId('restart').addEventListener('click', reset);
  byId('shuffle').addEventListener('click', () => {
    for (let end = numbers.length - 1; end > 0; end--) {
      const other = Math.floor(Math.random() * (end + 1));
      [numbers[other], numbers[end]] = [numbers[end], numbers[other]];
    }
    reset();
    announcement.textContent = 'Numbers shuffled.';
  });
  method.addEventListener('change', reset);
  speed.addEventListener('change', () => { if (timer !== null) start(); });
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) { stop(); paint(); }
  });
  reset();
})();

'use strict';

(() => {
  const byId = (id) => document.getElementById(id);
  const harmonies = {
    analogous: [-34, -17, 0, 17, 34],
    complementary: [0, 12, -12, 172, 188],
    triadic: [0, 112, 128, 232, 248]
  };
  const initial = ['#F2C197', '#FF7852', '#C9ADED', '#AF5477', '#4F324C'];
  const key = 'kr-palette-press-v1';
  const hexPattern = /^#[0-9a-f]{6}$/i;
  const harmony = byId('harmony');
  const board = byId('swatches');
  const output = byId('css-output');
  const status = byId('palette-status');
  let colors = initial.slice();
  let locked = initial.map(() => false);

  try {
    const saved = JSON.parse(localStorage.getItem(key));
    if (saved && Array.isArray(saved.colors) && saved.colors.length === 5 &&
        saved.colors.every(color => typeof color === 'string' && hexPattern.test(color)) &&
        Array.isArray(saved.locked) && saved.locked.length === 5 &&
        saved.locked.every(value => typeof value === 'boolean') &&
        Object.hasOwn(harmonies, saved.harmony)) {
      colors = saved.colors;
      locked = saved.locked;
      harmony.value = saved.harmony;
    }
  } catch { /* A fresh palette also works when storage is unavailable. */ }

  function toHex(h, s, l) {
    const hue = ((h % 360) + 360) % 360;
    const a = s * Math.min(l, 1 - l);
    function channel(n) {
      const k = (n + hue / 30) % 12;
      const rgb = l - a * Math.max(-1, Math.min(k - 3, 9 - k, 1));
      return Math.round(255 * rgb).toString(16).padStart(2, '0');
    }
    return ('#' + channel(0) + channel(8) + channel(4)).toUpperCase();
  }

  function ink(hex) {
    const rgb = hex.slice(1).match(/../g).map(value => {
      const c = parseInt(value, 16) / 255;
      return c <= .04045 ? c / 12.92 : ((c + .055) / 1.055) ** 2.4;
    });
    const luminance = .2126 * rgb[0] + .7152 * rgb[1] + .0722 * rgb[2];
    return luminance > .179 ? '#11110F' : '#FFFFFF';
  }

  const tiles = colors.map((_, index) => {
    const tile = document.createElement('div');
    tile.className = 'palette-tile';
    const code = document.createElement('span');
    code.className = 'tile-hex';
    const lock = document.createElement('button');
    lock.type = 'button';
    lock.className = 'lock-color';
    lock.addEventListener('click', () => {
      locked[index] = !locked[index];
      paint();
      status.textContent = 'Color ' + (index + 1) + (locked[index] ? ' locked.' : ' unlocked.');
    });
    tile.append(code, lock);
    board.append(tile);
    return { tile, code, lock };
  });

  function paint() {
    colors.forEach((color, i) => {
      const { tile, code, lock } = tiles[i];
      tile.style.backgroundColor = color;
      tile.style.color = ink(color);
      code.textContent = color;
      lock.textContent = locked[i] ? 'Locked' : 'Lock';
      lock.setAttribute('aria-label', (locked[i] ? 'Unlock' : 'Lock') + ' color ' + (i + 1));
      lock.setAttribute('aria-pressed', String(locked[i]));
    });
    output.value = ':root {\n' + colors.map((color, i) => '  --color-' + (i + 1) + ': ' + color + ';').join('\n') + '\n}';
    byId('generate').disabled = locked.every(Boolean);
    try { localStorage.setItem(key, JSON.stringify({ colors, locked, harmony: harmony.value })); } catch {}
  }

  function generate() {
    if (locked.every(Boolean)) {
      status.textContent = 'Unlock a color to shuffle the palette.';
      paint();
      return;
    }
    const hue = 360 * Math.random();
    const offsets = harmonies[harmony.value];
    const levels = [.8, .65, .71, .47, .29];
    colors = colors.map((color, i) => locked[i] ? color :
      toHex(hue + offsets[i], .46 + Math.random() * .33, levels[i]));
    paint();
    status.textContent = 'New palette ready.';
  }

  byId('generate').addEventListener('click', generate);
  harmony.addEventListener('change', generate);
  byId('copy-css').addEventListener('click', async () => {
    let copied = false;
    try { await navigator.clipboard.writeText(output.value); copied = true; }
    catch {
      output.focus();
      output.select();
      try { copied = document.execCommand('copy'); } catch {}
    }
    status.textContent = copied ? 'CSS copied.' : 'CSS selected. Press Ctrl+C or Command+C to copy.';
  });
  byId('download-css').addEventListener('click', () => {
    const file = new Blob([output.value + '\n'], { type: 'text/css;charset=utf-8' });
    const url = URL.createObjectURL(file);
    const link = document.createElement('a');
    link.href = url;
    link.download = 'palette.css';
    document.body.append(link);
    link.click();
    link.remove();
    setTimeout(() => URL.revokeObjectURL(url), 2000);
    status.textContent = 'Created palette.css.';
  });
  paint();
})();
